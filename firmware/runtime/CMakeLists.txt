cmake_minimum_required(VERSION 3.20)
option(DRONE_PICO2 "Build the physical Pico 2 neutral-only bench target" OFF)
if(DRONE_PICO2)
  set(PICOTOOL_GIT_BRANCH "2041936441b48a3cc53ae3da9e805229fe8f4e18" CACHE STRING "Pinned UF2 conversion tool" FORCE)
  set(PICO_BOARD pico2 CACHE STRING "Bench board" FORCE)
  set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Recorded Cortex-M33 target" FORCE)
  if(NOT DEFINED ENV{PICO_SDK_PATH})
    message(FATAL_ERROR "Set PICO_SDK_PATH to the pinned SDK checkout")
  endif()
  find_package(Git REQUIRED)
  execute_process(COMMAND ${GIT_EXECUTABLE} -C "$ENV{PICO_SDK_PATH}" rev-parse HEAD OUTPUT_VARIABLE SDK_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT SDK_REV STREQUAL "079c6f39023649b154152db30f1d781e884879bc")
    message(FATAL_ERROR "Use the pinned Pico SDK 2.3.1 commit")
  endif()
  include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(drone_runtime LANGUAGES C CXX ASM)
if(DRONE_PICO2 AND (NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "14.3.1"))
  message(FATAL_ERROR "Pico 2 build requires the recorded Arm GNU 14.3.Rel1 toolchain (GCC 14.3.1)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(drone_kernel STATIC src/runtime.cpp src/protocol.cpp)
target_include_directories(drone_kernel PUBLIC include)
target_compile_options(drone_kernel PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti -fstack-usage)
if(DRONE_PICO2)
  pico_sdk_init()
  option(DRONE_WATCHDOG_TEST "Intentionally hang after two seconds for disabled-load watchdog measurement" OFF)
  add_executable(drone_runtime src/pico.cpp)
  target_link_libraries(drone_runtime PRIVATE drone_kernel pico_stdlib pico_rand hardware_uart hardware_watchdog hardware_gpio)
  target_compile_definitions(drone_runtime PRIVATE DRONE_WATCHDOG_TEST=$<BOOL:${DRONE_WATCHDOG_TEST}> PICO_STACK_SIZE=4096 PICO_HEAP_SIZE=0)
  target_compile_options(drone_runtime PRIVATE -Wall -Wextra -Werror -fstack-usage "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>")
  target_link_options(drone_runtime PRIVATE "-Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/memory-budget.ld" -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
  pico_enable_stdio_usb(drone_runtime 0)
  pico_enable_stdio_uart(drone_runtime 0)
  pico_add_extra_outputs(drone_runtime)
else()
  add_executable(drone_runtime_host src/host.cpp)
  target_link_libraries(drone_runtime_host PRIVATE drone_kernel)
  target_compile_options(drone_runtime_host PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
  enable_testing()
  add_executable(drone_runtime_tests tests/runtime_test.cpp)
  target_link_libraries(drone_runtime_tests PRIVATE drone_kernel)
  add_test(NAME runtime_contract COMMAND drone_runtime_tests)
endif()
